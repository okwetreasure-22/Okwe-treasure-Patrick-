import tkinter as tk
from tkinter import messagebox
import ast
import operator

# Create the main window
root = tk.Tk()
root.title("My Calculator")
root.geometry("350x480")
root.resizable(False, False)
root.configure(bg="#1e293b")

# Calculator display
display = tk.Entry(
    root,
    font=("Arial", 24),
    justify="right",
    bd=10,
    relief="flat"
)
display.pack(fill="x", padx=15, pady=20, ipady=10)

# Allowed arithmetic operations
operations = {
    ast.Add: operator.add,
    ast.Sub: operator.sub,
    ast.Mult: operator.mul,
    ast.Div: operator.truediv,
    ast.Mod: operator.mod,
    ast.UAdd: operator.pos,
    ast.USub: operator.neg
}

def safe_calculate(expression):
    """Evaluate basic arithmetic without executing arbitrary Python code."""
    def evaluate(node):
        if isinstance(node, ast.Expression):
            return evaluate(node.body)

        if isinstance(node, ast.Constant) and type(node.value) in (int, float):
            return node.value

        if isinstance(node, ast.BinOp) and type(node.op) in operations:
            left = evaluate(node.left)
            right = evaluate(node.right)

            if isinstance(node.op, ast.Div) and right == 0:
                raise ZeroDivisionError

            if isinstance(node.op, ast.Mod) and right == 0:
                raise ZeroDivisionError

            return operations[type(node.op)](left, right)

        if isinstance(node, ast.UnaryOp) and type(node.op) in operations:
            return operations[type(node.op)](evaluate(node.operand))

        raise ValueError("Invalid expression")

    tree = ast.parse(expression, mode="eval")
    return evaluate(tree)

def button_click(value):
    display.insert(tk.END, value)

def clear_display():
    display.delete(0, tk.END)

def delete_last():
    current = display.get()
    display.delete(0, tk.END)
    display.insert(0, current[:-1])

def calculate():
    expression = display.get().strip()

    if not expression:
        return

    try:
        result = safe_calculate(expression)

        if isinstance(result, float):
            if result.is_integer():
                result = int(result)
            else:
                result = round(result, 10)

        display.delete(0, tk.END)
        display.insert(0, str(result))

    except ZeroDivisionError:
        messagebox.showerror("Error", "Cannot divide by zero!")
        clear_display()

    except (ValueError, SyntaxError, OverflowError):
        messagebox.showerror("Error", "Invalid calculation!")
        clear_display()

# Button layout
buttons = [
    ("7", 0, 0), ("8", 0, 1), ("9", 0, 2), ("/", 0, 3),
    ("4", 1, 0), ("5", 1, 1), ("6", 1, 2), ("*", 1, 3),
    ("1", 2, 0), ("2", 2, 1), ("3", 2, 2), ("-", 2, 3),
    ("0", 3, 0), (".", 3, 1), ("%", 3, 2), ("+", 3, 3)
]

# Create the buttons
frame = tk.Frame(root, bg="#1e293b")
frame.pack(padx=10, fill="both", expand=True)

for text, row, col in buttons:
    button = tk.Button(
        frame,
        text=text,
        font=("Arial", 18, "bold"),
        bg="#334155",
        fg="white",
        activebackground="#475569",
        activeforeground="white",
        relief="flat",
        command=lambda value=text: button_click(value)
    )
    button.grid(
        row=row,
        column=col,
        sticky="nsew",
        padx=5,
        pady=5,
        ipady=10
    )

# Clear, delete and equals buttons
tk.Button(
    frame,
    text="CLEAR",
    font=("Arial", 14, "bold"),
    bg="#ef4444",
    fg="white",
    command=clear_display
).grid(row=4, column=0, columnspan=2, sticky="nsew", padx=5, pady=5)

tk.Button(
    frame,
    text="DEL",
    font=("Arial", 14, "bold"),
    bg="#64748b",
    fg="white",
    command=delete_last
).grid(row=4, column=2, sticky="nsew", padx=5, pady=5)

tk.Button(
    frame,
    text="=",
    font=("Arial", 18, "bold"),
    bg="#22c55e",
    fg="black",
    command=calculate
).grid(row=4, column=3, sticky="nsew", padx=5, pady=5)

# Make the grid responsive within the window
for col in range(4):
    frame.grid_columnconfigure(col, weight=1)

for row in range(5):
    frame.grid_rowconfigure(row, weight=1)

# Run the application
root.mainloop()
